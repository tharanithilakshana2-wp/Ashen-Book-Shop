-- ============================================================================
-- 1. IDENTITY / TENANCY
-- ============================================================================

create table if not exists businesses (
  id uuid primary key default uuid_generate_v4(),
  name text not null,
  logo_url text,
  address text,
  phone text,
  email text,
  registration_no text,
  currency_code text not null default 'LKR',
  currency_symbol text not null default 'Rs',
  timezone text not null default 'Asia/Colombo',
  default_language text not null default 'en' check (default_language in ('en','si')),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists branches (
  id uuid primary key default uuid_generate_v4(),
  business_id uuid not null references businesses(id) on delete cascade,
  name text not null,
  address text,
  phone text,
  is_active boolean not null default true,
  created_at timestamptz not null default now()
);
create index if not exists idx_branches_business on branches(business_id);

-- ============================================================================
-- 2. AUTH / RBAC

create table if not exists roles (
  id uuid primary key default uuid_generate_v4(),
  business_id uuid references businesses(id) on delete cascade,
  name text not null,               -- 'super_admin','owner','manager','cashier','inventory_manager'
  is_system boolean not null default false,
  unique(business_id, name)
);

create table if not exists permissions (
  code text primary key,            -- e.g. 'pos.sell', 'reports.view', 'inventory.adjust'
  description text
);

create table if not exists user_permissions (
  user_id uuid not null references auth.users(id) on delete cascade,
  permission_code text not null references permissions(code) on delete cascade,
  granted boolean not null default true,
  primary key (user_id, permission_code)
);

create table if not exists users (
  id uuid primary key references auth.users(id) on delete cascade,
  business_id uuid not null references businesses(id) on delete cascade,
  branch_id uuid references branches(id),
  role_id uuid not null references roles(id),
  full_name text not null,
  phone text,
  avatar_url text,
  is_active boolean not null default true,
  created_by uuid references auth.users(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
create index if not exists idx_users_business on users(business_id);

  business_id uuid not null references businesses(id) on delete cascade,
  name text not null,

create table if not exists products (
  id uuid primary key default uuid_generate_v4(),
  business_id uuid not null references businesses(id) on delete cascade,
  branch_id uuid references branches(id),
  name_en text not null,
  name_si text,
  description text,
  category_id uuid references categories(id),
  brand_id uuid references brands(id),
  barcode text,
  sku text,
  image_url text,
  purchase_price numeric(12,2) not null default 0,
  selling_price numeric(12,2) not null default 0,
  wholesale_price numeric(12,2),
  min_stock numeric(12,2) not null default 0,
  unit text not null default 'pcs',   -- pcs, kg, L, box...

  updated_at timestamptz not null default now(),

create index if not exists idx_products_barcode on products(barcode);
create index if not exists idx_products_name on products using gin (to_tsvector('simple', name_en));

create table if not exists product_barcodes (
  id uuid primary key default uuid_generate_v4(),
  product_id uuid not null references products(id) on delete cascade,
  barcode text not null,
  format text not null default 'EAN-13' check (format in ('EAN-13','EAN-8','UPC','CODE128')),

  created_at timestamptz not null default now(),

);
create index if not exists idx_pb_barcode on product_barcodes(barcode);

create table if not exists inventory (

  quantity numeric(14,3) not null default 0 check (quantity >= 0),

  id uuid primary key default uuid_generate_v4(),
  business_id uuid not null references businesses(id),
  product_id uuid not null references products(id),

create table if not exists sales (
  id uuid primary key default uuid_generate_v4(),
  business_id uuid not null references businesses(id) on delete cascade,
  branch_id uuid references branches(id),
  invoice_no text not null,                 -- unique per business
  customer_id uuid references customers(id),  -- null = walk-in
  cashier_id uuid not null references auth.users(id),
  shift_id uuid,                            -- cashier_shifts (§17)
  sale_date timestamptz not null default now(),
  subtotal numeric(14,2) not null default 0,
  item_discount_total numeric(14,2) not null default 0,
  invoice_discount numeric(14,2) not null default 0,
  tax_total numeric(14,2) not null default 0,
  total numeric(14,2) not null,
  payment_status text not null default 'paid' check (payment_status in ('paid','credit','partial','refunded')),
  sync_status text not null default 'synced' check (sync_status in ('synced','pending','failed')),
  client_txn_id uuid,                       -- idempotency key from offline POS (§15)
  notes text,
  created_at timestamptz not null default now(),
  unique(business_id, invoice_no),
  unique(business_id, client_txn_id)
);

create or replace function current_business_id() returns uuid
language sql stable as $$
  select business_id from users where id = auth.uid()
$$;

-- Users see only their own business data
create policy tenant_isolation on users
  for all using (business_id = current_business_id());

create policy tenant_products on products
  for all using (business_id = current_business_id());

create policy tenant_sales on sales
  for all using (business_id = current_business_id());

create policy tenant_customers on customers
  for all using (business_id = current_business_id());

create policy tenant_movements on inventory_movements
  for select using (business_id = current_business_id());

-- ============================================================================
-- ATOMIC SALE TRANSACTION (example function - Stage 4 wires the API to this)
-- ============================================================================

create or replace function process_sale(
  p_business uuid, p_branch uuid, p_cashier uuid, p_shift uuid,
  p_customer uuid, p_client_txn uuid, p_items jsonb, p_payments jsonb
) returns uuid
language plpgsql security definer as $$
declare
  v_sale uuid;
  v_item jsonb;
  v_inv_no text;
  v_wac numeric;
begin
  -- Idempotency: offline retries must not duplicate (§15)
  select id into v_sale from sales where business_id = p_business and client_txn_id = p_client_txn;
  if v_sale is not null then return v_sale; end if;

  select invoice_no into v_inv_no from sales where business_id = p_business
  order by created_at desc limit 1;
  v_inv_no := coalesce(v_inv_no, 'INV-0');
  v_inv_no := 'INV-' || (substring(v_inv_no from 5)::int + 1);

  insert into sales (business_id, branch_id, invoice_no, customer_id, cashier_id, shift_id, total)
  values (p_business, p_branch, v_inv_no, p_customer, p_cashier, p_shift, 0)
  returning id into v_sale;

  for v_item in select * from jsonb_array_elements(p_items)
  loop
    select weighted_avg_cost into v_wac from products where id = (v_item->>'product_id')::uuid;
    -- weighted average cost update happens on purchase receipt (Stage 6)
    insert into sale_items (sale_id, product_id, quantity, unit_price, unit_cost, line_total)
    values (v_sale, (v_item->>'product_id')::uuid, (v_item->>'qty')::numeric,
            (v_item->>'price')::numeric, v_wac, (v_item->>'line_total')::numeric);

    insert into inventory_movements (business_id, product_id, movement_type, quantity_change,
      quantity_before, quantity_after, reference_id, created_by)
    select p_business, (v_item->>'product_id')::uuid, 'sale_out', -(v_item->>'qty')::numeric,
      i.quantity, i.quantity - (v_item->>'qty')::numeric, v_sale, p_cashier
    from inventory i where i.product_id = (v_item->>'product_id')::uuid;

    update inventory set quantity = quantity - (v_item->>'qty')::numeric,
      updated_at = now()
    where product_id = (v_item->>'product_id')::uuid;
  end loop;

  insert into payments (sale_id, payment_method, amount, created_by)
  select v_sale, p->>'method', (p->>'amount')::numeric, p_cashier
  from jsonb_array_elements(p_payments) p;

  return v_sale;
end $$;
